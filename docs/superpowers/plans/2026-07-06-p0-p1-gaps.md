```python
"""流式输出工具：idle timeout 保护"""
import asyncio


async def stream_with_idle_timeout(aiter, timeout: float = 30.0):
    """异步迭代器 idle timeout 包装。

    每个 chunk 之间设 idle 超时（不是总超时）。
    超过 timeout 秒未收到下一个 chunk 则抛出 TimeoutError。

    Args:
        aiter: 异步迭代器（如 llm.astream() 的返回值）
        timeout: 两个 chunk 之间允许的最大间隔秒数
    """
    while True:
        try:
            chunk = await asyncio.wait_for(aiter.__anext__(), timeout)
            yield chunk
        except StopAsyncIteration:
            break
        except asyncio.TimeoutError:
            raise TimeoutError(f"LLM 无响应超过 {timeout} 秒")
```

```python
import ast


def _extract_search_sources(output: str) -> list[dict]:
    """从 TavilySearchResults 的字符串输出中提取来源。

    web_search 工具返回 str(results)，需要解析回 list 提取 url + title。
    """
    try:
        results = ast.literal_eval(output)
    except (ValueError, SyntaxError):
        return []
    sources = []
    if isinstance(results, list):
        for r in results:
            if isinstance(r, dict) and r.get("url"):
                title = r.get("title", "")
                sources.append({
                    "source": f"{title} - {r['url']}" if title else r["url"],
                    "score": 0,
                })
    return sources
```

```python
async def _agent_stream(prompt: str):
    """AgentExecutor 流式：工具调用 status + LLM token + web_search sources

    yield 字典事件：
      {"type": "status", "content": "..."}  — web_search 开始时
      {"type": "token", "content": "..."}   — LLM 流式 token
      {"type": "sources", "content": [...]} — 最终合并的 web_search 来源
    """
    from langchain_core.prompts import ChatPromptTemplate
    from langchain.agents import create_tool_calling_agent, AgentExecutor

    llm = get_llm(temperature=0.3)
    tools = [web_search, calculator]
    agent_prompt = ChatPromptTemplate.from_messages([
        ("system", "你是一个知识库问答助手。请根据提供的上下文回答问题。如果需要最新信息可以搜索，需要计算可以用计算器。"),
        ("human", "{input}"),
        ("placeholder", "{agent_scratchpad}"),
    ])
    agent = create_tool_calling_agent(llm=llm, tools=tools, prompt=agent_prompt)
    agent_executor = AgentExecutor(agent=agent, tools=tools, max_iterations=3, verbose=False)

    collected_sources = []

    async for event in agent_executor.astream_events({"input": prompt}, version="v2"):
        etype = event["event"]

        # 工具开始：仅 web_search 发 status（有网络延迟，用户可感知）
        if etype == "on_tool_start" and event["name"] == "web_search":
            query = event["data"].get("input", {}).get("query", "")
            if isinstance(query, str) and query:
                yield {"type": "status", "content": f"正在搜索：{query}"}

        # 工具结束：仅 web_search 收集来源（calculator 无来源语义）
        elif etype == "on_tool_end" and event["name"] == "web_search":
            output = event["data"].get("output", "")
            if isinstance(output, str):
                collected_sources.extend(_extract_search_sources(output))

        # LLM 流式 token
        elif etype == "on_chat_model_stream":
            token = event["data"]["chunk"].content
            if token:
                yield {"type": "token", "content": token}

    # 最终合并来源
    if collected_sources:
        yield {"type": "sources", "content": collected_sources}
```

```python
"""记忆路由：GET /api/v1/memory/recall — 学习历程"""
from fastapi import APIRouter, Depends, Query
from sqlalchemy.orm import Session

from app.deps import get_db
from app.models.schemas import EpisodicEventResponse
from memory.episodic import get_recent_events

router = APIRouter(prefix="/api/v1/memory", tags=["memory"])


@router.get("/recall", response_model=list[EpisodicEventResponse])
def recall_memory(
    user_id: str = Query(...),
    limit: int = Query(20, le=100),
    db: Session = Depends(get_db),
):
    """获取用户的学习历程（情景记忆事件列表）"""
    events = get_recent_events(db, user_id, limit=limit)
    return [
        {
            "id": e.id,
            "event_type": e.event_type,
            "content": e.content,
            "question_text": e.question_text,
            "created_at": e.created_at.isoformat() if e.created_at else None,
        }
        for e in events
    ]
```
